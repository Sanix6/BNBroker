import uuid
from django.conf import settings
from django.db import models

class AMLCheck(models.Model):
    class Kind(models.TextChoices):
        ADDRESS = "address", "Проверка адреса"
        TRANSACTION = "transaction", "Проверка TxID"
    class Status(models.TextChoices):
        PENDING = "pending", "Ожидает"
        COMPLETED = "completed", "Завершена"
        FAILED = "failed", "Ошибка"
    class Decision(models.TextChoices):
        PENDING = "pending", "Не принято"
        APPROVED = "approved", "Одобрено"
        REVIEW = "manual_review", "Ручная проверка"
        BLOCKED = "blocked", "Заблокировано"
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    user = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.PROTECT, null=True, blank=True, related_name="crypto_aml_checks")
    asset_network = models.ForeignKey("crypto.CryptoNetwork", on_delete=models.PROTECT, related_name="aml_checks")
    crypto_address = models.ForeignKey("crypto.CryptoAddress", on_delete=models.SET_NULL, null=True, blank=True, related_name="aml_checks")
    kind = models.CharField(max_length=20, choices=Kind.choices)
    address = models.CharField(max_length=255, blank=True)
    txid = models.CharField(max_length=255, blank=True)
    provider = models.CharField(max_length=20, default="ranex", editable=False)
    provider_check_id = models.CharField(max_length=128, blank=True, db_index=True)
    status = models.CharField(max_length=20, choices=Status.choices, default=Status.PENDING, db_index=True)
    decision = models.CharField(max_length=20, choices=Decision.choices, default=Decision.PENDING, db_index=True)
    risk_score = models.DecimalField(max_digits=8, decimal_places=2, null=True, blank=True)
    risk_category = models.CharField(max_length=100, blank=True)
    provider_payload = models.JSONField(default=dict, blank=True, help_text="Очищенный ответ Ranex без токенов и лишних ПД")
    error_message = models.TextField(blank=True)
    screened_at = models.DateTimeField(null=True, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)


    class Meta:
        verbose_name = "AML-проверка"
        verbose_name_plural = "AML-проверки"
        ordering = ["-created_at"]
        indexes = [models.Index(fields=["kind","status","created_at"], name="aml_kind_status_created_idx")]
        constraints = [models.CheckConstraint(condition=(models.Q(kind="address", address__gt="", txid="") | models.Q(kind="transaction", txid__gt="", address="")), name="aml_address_or_txid_required")]
        
    def __str__(self): return f"{self.kind}: {self.address or self.txid}"

from rest_framework import serializers
from apps.compliance.models import AMLCheck
from apps.crypto.models import CryptoNetwork


class AMLAddressCheckSerializer(serializers.Serializer):
    asset_network_id = serializers.PrimaryKeyRelatedField(
        queryset=CryptoNetwork.objects.filter(
            is_active=True,
            asset__is_active=True,
            network__is_active=True,
        ),
        source="asset_network",
    )
    address = serializers.CharField(
        max_length=255,
        trim_whitespace=True,
    )

    def validate(self, attrs):
        asset_network = attrs["asset_network"]
        network = asset_network.network

        if not network.ranex_network_code:
            raise serializers.ValidationError({
                "asset_network_id": "AML-проверка этой сети не поддерживается."
            })

        from apps.integrations.ranex.constants import RanexNetwork

        if network.ranex_network_code not in RanexNetwork.SUPPORTED:
            raise serializers.ValidationError({
                "asset_network_id": "Сеть не поддерживается Ranex."
            })

        return attrs


class AMLCheckSerializer(serializers.ModelSerializer):
    asset_network_id = serializers.IntegerField(
        read_only=True,
    )

    class Meta:
        model = AMLCheck
        fields = [
            "id",
            "asset_network_id",
            "kind",
            "address",
            "provider",
            "status",
            "decision",
            "risk_score",
            "risk_category",
            "created_at",
            "screened_at",
        ]
        read_only_fields = fields

from decimal import Decimal, InvalidOperation
from django.utils import timezone
from apps.compliance.models import AMLCheck
from apps.integrations.ranex.client import RanexClient
from apps.integrations.ranex.constants import RanexNetwork


def screen_crypto_address(*, asset_network, address, user=None, crypto_address=None):
    network = (asset_network.network.ranex_network_code or asset_network.network.code).strip().lower()
    if network not in RanexNetwork.SUPPORTED:
        raise ValueError(f"Сеть {network} не поддерживается клиентом Ranex")
    if not asset_network.is_active or not asset_network.aml_screening_required:
        raise ValueError("AML-проверка отключена для данной пары")
    address = (address or '').strip()
    if not address or len(address) > 255:
        raise ValueError("Укажите корректный адрес")
    if crypto_address and (crypto_address.asset_network_id != asset_network.pk or crypto_address.address != address):
        raise ValueError("Криптоадрес не соответствует выбранной сети")
    check = AMLCheck.objects.create(
        user=user, asset_network=asset_network, crypto_address=crypto_address,
        kind=AMLCheck.Kind.ADDRESS, address=address,
        status=AMLCheck.Status.PENDING, decision=AMLCheck.Decision.PENDING,
    )
    try:
        response = RanexClient().screen_address(network=network, address=address)
        risk_info = (response.get('risk') or {}).get('riskInfo') or {}
        score = risk_info.get('risk')
        try:
            score = Decimal(str(score)) if score is not None else None
        except (InvalidOperation, TypeError, ValueError):
            score = None
        reported = risk_info.get('reported') or {}
        check.risk_score = score
        check.risk_category = str(reported.get('name') or '')[:100]
        check.provider_check_id = str(response.get('id') or response.get('screeningId') or '')[:128]
        check.provider_payload = {'network': network, 'risk_category': check.risk_category}
        check.status = AMLCheck.Status.COMPLETED
        check.decision = AMLCheck.Decision.REVIEW
        check.screened_at = timezone.now()
        check.save()
        return check
    except Exception as exc:
        check.status = AMLCheck.Status.FAILED
        check.error_message = exc.__class__.__name__
        check.save(update_fields=['status','error_message','updated_at'])
        raise

import logging
from decimal import Decimal, InvalidOperation
from django.conf import settings
from django.utils import timezone
from rest_framework import generics, permissions, status
from rest_framework.response import Response
from rest_framework.views import APIView
from drf_spectacular.utils import extend_schema

from apps.compliance.models import AMLCheck
from apps.compliance.serializers import (
    AMLAddressCheckSerializer,
    AMLCheckSerializer,
)
from apps.integrations.ranex.client import RanexClient
from apps.integrations.ranex.exceptions import RanexError

logger = logging.getLogger(__name__)


class AMLAddressCheckView(APIView):
    permission_classes = [permissions.IsAuthenticated]

    @extend_schema(
        request=AMLAddressCheckSerializer,
        responses={201: AMLCheckSerializer},
        tags=["AML"],
    )
    def post(self, request):
        serializer = AMLAddressCheckSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        asset_network = serializer.validated_data["asset_network"]
        address = serializer.validated_data["address"]

        check = AMLCheck.objects.create(
            user=request.user,
            asset_network=asset_network,
            kind=AMLCheck.Kind.ADDRESS,
            address=address,
            status=AMLCheck.Status.PENDING,
            decision=AMLCheck.Decision.PENDING,
        )

        try:
            result = RanexClient().screen_address(
                network=asset_network.network.ranex_network_code,
                address=address,
            )

            # Проверяем вложенную структуру Ranex перед завершением проверки.
            if not isinstance(result, dict):
                raise ValueError("Invalid Ranex response")
            risk = result.get("risk")
            if not isinstance(risk, dict):
                raise ValueError("Missing Ranex risk data")
            errors = risk.get("errors")
            if not isinstance(errors, list) or errors:
                raise ValueError("Ranex risk errors or invalid errors field")
            risk_info = risk.get("riskInfo")
            if not isinstance(risk_info, dict):
                raise ValueError("Missing Ranex riskInfo")
            raw_score = risk_info.get("risk")
            if isinstance(raw_score, bool) or not isinstance(raw_score, (int, float, Decimal)):
                raise ValueError("Invalid Ranex risk score type")
            try:
                risk_score = Decimal(str(raw_score))
            except InvalidOperation as exc:
                raise ValueError("Invalid Ranex risk score") from exc
            if not risk_score.is_finite() or risk_score < 0 or risk_score >= Decimal("1000000"):
                raise ValueError("Ranex risk score out of supported range")

            # Не сохраняем непроверенный полный ответ провайдера:
            # он может содержать лишние или чувствительные данные.
            # Не одобряем адрес без подтверждённой схемы Ranex.
            check.status = AMLCheck.Status.COMPLETED
            check.decision = AMLCheck.Decision.REVIEW
            check.risk_score = risk_score
            check.screened_at = timezone.now()
            check.save(update_fields=[
                "status",
                "decision",
                "risk_score",
                "screened_at",
                "updated_at",
            ])

            logger.info(
                "AML screening completed; manual review required. check_id=%s",
                check.id,
            )

            return Response(
                AMLCheckSerializer(check).data,
                status=status.HTTP_201_CREATED,
            )

        except (RanexError, ValueError):
            logger.exception(
                "Ranex AML screening failed. check_id=%s",
                check.id,
            )

            check.status = AMLCheck.Status.FAILED
            check.decision = AMLCheck.Decision.PENDING
            check.error_message = "Provider screening failed or returned invalid data"
            check.save(update_fields=[
                "status",
                "decision",
                "error_message",
                "updated_at",
            ])

            return Response(
                {
                    "id": str(check.id),
                    "status": "failed",
                    "decision": "pending",
                    "detail": "AML-провайдер временно недоступен или отклонил запрос.",
                },
                status=status.HTTP_502_BAD_GATEWAY,
            )

class ExchangeOrderAdminForm(forms.ModelForm):
    class Meta:
        model = ExchangeOrder
        fields = '__all__'

    def clean(self):
        data = super().clean()
        if not self.instance.pk:
            return data
        old = ExchangeOrder.objects.get(pk=self.instance.pk)
        new_status = data.get('status', old.status)
        transitions = {
            'awaiting_payment': {'payment_review', 'cancelled', 'rejected'},
            'payment_review': {'paid', 'cancelled', 'rejected'},
            'paid': {'processing'},
            'processing': {'completed', 'rejected'},
            'completed': set(), 'cancelled': set(), 'rejected': set(),
        }
        if new_status != old.status and new_status not in transitions.get(old.status, set()):
            raise ValidationError('Недопустимый переход статуса заявки')
        if new_status == 'completed' and not data.get('txid'):
            raise ValidationError('Для завершения требуется TxID')
        return data

"""Read-only public catalog of operator-enabled exchange pairs."""
from django.db.models import Q
from drf_spectacular.utils import extend_schema, OpenApiParameter
from rest_framework.permissions import AllowAny
from rest_framework.response import Response
from rest_framework.views import APIView
from rest_framework.exceptions import ValidationError
from .models import ExchangeDirection


def active_directions():
    return (ExchangeDirection.objects.filter(is_active=True)
        .filter(
            Q(from_kind='fiat') |
            Q(from_asset_network__is_active=True,
              from_asset_network__asset__is_active=True,
              from_asset_network__network__is_active=True,
              from_asset_network__deposit_enabled=True)
        ).filter(
            Q(to_kind='fiat') |
            Q(to_asset_network__is_active=True,
              to_asset_network__asset__is_active=True,
              to_asset_network__network__is_active=True,
              to_asset_network__withdrawal_enabled=True)
        ).select_related(
            'from_asset_network__asset', 'from_asset_network__network',
            'to_asset_network__asset', 'to_asset_network__network',
        ).order_by('id'))


def node(direction, side):
    kind = getattr(direction, f'{side}_kind')
    if kind == 'fiat':
        return {'id': 'fiat:RUB', 'kind': 'fiat', 'currency': 'RUB',
                'asset_network_id': None, 'network': None}
    item = getattr(direction, f'{side}_asset_network')
    return {'id': f'crypto:{item.pk}', 'kind': 'crypto',
            'currency': item.asset.symbol, 'asset_network_id': item.pk,
            'network': item.network.code}


def serialize(direction):
    left, right = node(direction, 'from'), node(direction, 'to')
    return {'id': direction.pk, 'from': left, 'to': right,
            'from_currency': left['currency'], 'to_currency': right['currency'],
            'is_active': True}


class ExchangeAssetsView(APIView):
    permission_classes = [AllowAny]

    @extend_schema(tags=['Exchange'], responses={200: dict})
    def get(self, request):
        directions = list(active_directions())
        from_nodes, to_nodes = {}, {}
        for direction in directions:
            left, right = node(direction, 'from'), node(direction, 'to')
            from_nodes[left['id']] = left
            to_nodes[right['id']] = right
        return Response({'from': sorted(from_nodes.values(), key=lambda x: x['id']),
                         'to': sorted(to_nodes.values(), key=lambda x: x['id'])})


class ExchangeDirectionsView(APIView):
    permission_classes = [AllowAny]

    @extend_schema(
        tags=['Exchange'],
        parameters=[
            OpenApiParameter('from', str, description='Node ID, e.g. fiat:RUB or crypto:4'),
            OpenApiParameter('to', str, description='Node ID, e.g. fiat:RUB or crypto:4'),
        ],
        responses={200: dict},
    )
    def get(self, request):
        from_id = request.query_params.get('from')
        to_id = request.query_params.get('to')
        for name, value in [('from', from_id), ('to', to_id)]:
            if value and value != 'fiat:RUB':
                if not value.startswith('crypto:') or not value[7:].isdigit() or int(value[7:]) < 1:
                    raise ValidationError({name: 'Используйте fiat:RUB или crypto:<asset_network_id>.'})
        directions = active_directions()
        if from_id:
            if from_id == 'fiat:RUB':
                directions = directions.filter(from_kind='fiat')
            else:
                directions = directions.filter(from_kind='crypto', from_asset_network_id=int(from_id[7:]))
        if to_id:
            if to_id == 'fiat:RUB':
                directions = directions.filter(to_kind='fiat')
            else:
                directions = directions.filter(to_kind='crypto', to_asset_network_id=int(to_id[7:]))
        return Response([serialize(direction) for direction in directions])

class ExchangeDirection(models.Model):
    """Operator-managed directed pair. RUB is the currently supported fiat endpoint."""
    class Kind(models.TextChoices):
        FIAT = 'fiat', 'Фиат'
        CRYPTO = 'crypto', 'Криптовалюта'

    from_kind = models.CharField(max_length=6, choices=Kind.choices, default=Kind.FIAT)
    from_asset_network = models.ForeignKey(
        'crypto.CryptoNetwork', on_delete=models.PROTECT, null=True, blank=True,
        related_name='exchange_directions_from', verbose_name='Отдаёт: сеть криптовалюты',
    )
    to_kind = models.CharField(max_length=6, choices=Kind.choices, default=Kind.CRYPTO)
    to_asset_network = models.ForeignKey(
        'crypto.CryptoNetwork', on_delete=models.PROTECT, null=True, blank=True,
        related_name='exchange_directions_to', verbose_name='Получает: сеть криптовалюты',
    )
    is_active = models.BooleanField(default=False, db_index=True, verbose_name='Доступно')
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        verbose_name = 'Направление обмена'
        verbose_name_plural = 'Направления обмена'
        ordering = ('id',)
        constraints = [
            models.UniqueConstraint(
                fields=['from_kind', 'from_asset_network', 'to_kind', 'to_asset_network'],
                name='exchange_direction_unique_pair',
                nulls_distinct=False,
            ),
        ]

    def clean(self):
        from django.core.exceptions import ValidationError
        super().clean()
        errors = {}
        for side in ('from', 'to'):
            kind = getattr(self, f'{side}_kind')
            network_id = getattr(self, f'{side}_asset_network_id')
            if kind == self.Kind.FIAT and network_id is not None:
                errors[f'{side}_asset_network'] = 'Для RUB сеть должна быть пустой.'
            if kind == self.Kind.CRYPTO and network_id is None:
                errors[f'{side}_asset_network'] = 'Выберите криптовалюту и сеть.'
        if self.from_kind == self.to_kind == self.Kind.FIAT:
            errors['to_kind'] = 'RUB → RUB не является направлением обмена.'
        if (self.from_kind == self.to_kind == self.Kind.CRYPTO and
                self.from_asset_network_id is not None and
                self.from_asset_network_id == self.to_asset_network_id):
            errors['to_asset_network'] = 'Нельзя обменивать актив на самого себя в той же сети.'
        if errors:
            raise ValidationError(errors)

    def __str__(self):
        left = 'RUB' if self.from_kind == self.Kind.FIAT else str(self.from_asset_network)
        right = 'RUB' if self.to_kind == self.Kind.FIAT else str(self.to_asset_network)
        return f'{left} → {right}'

@admin.register(ExchangeRate)

    list_display = ('direction', 'rate', 'is_active', 'updated_at', 'updated_by')
    list_display_links = ('direction',)
    list_editable = ('rate', 'is_active')
    list_filter = ('is_active', 'direction__from_kind', 'direction__to_kind')
    search_fields = (
        'direction__from_asset_network__asset__symbol',
        'direction__to_asset_network__asset__symbol',
        'direction__from_asset_network__network__code',
        'direction__to_asset_network__network__code',
    )
    readonly_fields = ('updated_at', 'created_at', 'updated_by')
    list_select_related = ('direction', 'updated_by')
    list_per_page = 40
    ordering = ('direction_id',)
    save_on_top = True

    def get_readonly_fields(self, request, obj=None):
        fields = list(super().get_readonly_fields(request, obj))
        if obj is not None:
            fields.append('direction')
        return tuple(fields)

    def has_delete_permission(self, request, obj=None):
        return False

    @transaction.atomic
    def save_model(self, request, obj, form, change):
        previous = None
        if change:
            previous = (
                ExchangeRate.objects.select_for_update()
                .filter(pk=obj.pk)
                .values('rate', 'is_active')
                .first()
            )

        obj.updated_by = request.user
        obj.full_clean()
        super().save_model(request, obj, form, change)

        if (previous is None
                or previous['rate'] != obj.rate
                or previous['is_active'] != obj.is_active):
            ExchangeRateChange.objects.create(
                direction=obj.direction,
                old_rate=previous['rate'] if previous else None,
                new_rate=obj.rate,
                old_active=previous['is_active'] if previous else None,
                new_active=obj.is_active,
                changed_by=request.user,
            )

                          PaymentSerializer, ReceiptUploadSerializer, ReceiptSerializer, ReportSerializer)

class DirectionsView(APIView):
    """Public fiat/crypto directions based on enabled network operations."""
    permission_classes = [AllowAny]

    def get(self, request):
        from django.db.models import Q
        from apps.crypto.models import CryptoNetwork

        networks = CryptoNetwork.objects.filter(
            Q(deposit_enabled=True) | Q(withdrawal_enabled=True),
            is_active=True, asset__is_active=True, network__is_active=True,
        ).select_related('asset', 'network').order_by(
            'asset__symbol', 'network__code', 'pk'
        )
        result = []
        for item in networks:
            common = {
                'asset_network_id': item.pk,
                'network': item.network.code,
            }
            if item.withdrawal_enabled:
                result.append({**common, 'id': f'{item.pk}:buy',
                               'from_currency': 'RUB', 'to_currency': item.asset.symbol})
            if item.deposit_enabled:
                result.append({**common, 'id': f'{item.pk}:sell',
                               'from_currency': item.asset.symbol, 'to_currency': 'RUB'})
        return Response(result)

class OrderCreateView(APIView):
    permission_classes = [IsAuthenticated]

    @extend_schema(request=OrderCreateSerializer, responses={201: OrderSerializer}, tags=['Exchange'])
    def post(self, request):
        s = OrderCreateSerializer(data=request.data)
        s.is_valid(raise_exception=True)
        d = s.validated_data
        if not KYCVerification.objects.filter(user=request.user, status=KYCVerification.Status.APPROVED).exists():
            raise PermissionDenied('Необходимо пройти KYC')
        with transaction.atomic():
            quote = get_object_or_404(ExchangeQuote.objects.select_for_update(), pk=d['quote_id'], user=request.user)
            if not quote.is_valid:
                raise ValidationError({'quote_id': 'Срок действия курса истёк'})
            if ExchangeOrder.objects.filter(quote=quote).exists():
                raise ValidationError({'quote_id': 'Котировка уже использована'})
            if not quote.asset_network.is_active or not quote.asset_network.network.is_active or not quote.asset_network.asset.is_active:
                raise ValidationError('Валюта или сеть отключена')
            if quote.side == 'buy' and not quote.asset_network.withdrawal_enabled:
                raise ValidationError('Вывод по этой сети отключён')
            if quote.side == 'sell' and not quote.asset_network.deposit_enabled:
                raise ValidationError('Приём по этой сети отключён')
            check = None
            if quote.asset_network.aml_screening_required:
                if not d.get('aml_check_id'):
                    raise ValidationError({'aml_check_id': 'Требуется проверка кошелька Ranex'})
                check = get_object_or_404(AMLCheck, pk=d['aml_check_id'], user=request.user,
                    asset_network=quote.asset_network, kind=AMLCheck.Kind.ADDRESS,
                    address=d['wallet_address'], status=AMLCheck.Status.COMPLETED,
                    decision=AMLCheck.Decision.APPROVED)
            order = ExchangeOrder.objects.create(user=request.user, quote=quote,
                wallet_address=d['wallet_address'], wallet_memo=d.get('wallet_memo', ''), aml_check=check)
        return Response(OrderSerializer(order).data, status=status.HTTP_201_CREATED)

class ReceiptUploadView(APIView):
    permission_classes = [IsAuthenticated]
    parser_classes = [MultiPartParser, FormParser]

    @extend_schema(request=ReceiptUploadSerializer, responses={201: ReceiptSerializer}, tags=['Exchange'])
    def post(self, request, pk):
        s = ReceiptUploadSerializer(data=request.data)
        s.is_valid(raise_exception=True)
        f = s.validated_data['file']
        if f.size > 5 * 1024 * 1024:
            raise ValidationError({'file': 'Максимальный размер 5 МБ'})
        if f.content_type not in {'image/jpeg', 'image/png', 'application/pdf'}:
            raise ValidationError({'file': 'Разрешены только JPEG, PNG, PDF'})
        with transaction.atomic():
            order = get_object_or_404(ExchangeOrder.objects.select_for_update(), pk=pk, user=request.user)
            if order.status not in {ExchangeOrder.Status.AWAITING_PAYMENT, ExchangeOrder.Status.PAYMENT_REVIEW}:
                raise ValidationError('Для этой заявки загрузка квитанции недоступна')
            receipt = PaymentReceipt.objects.create(order=order, file=f)
            order.status = ExchangeOrder.Status.PAYMENT_REVIEW
            order.save(update_fields=['status', 'updated_at'])
        return Response(ReceiptSerializer(receipt).data, status=status.HTTP_201_CREATED)

import json
import logging
from urllib.parse import quote

import requests
from django.conf import settings

from apps.integrations.ranex.constants import (
    RanexEndpoints,
    RanexNetwork,
)
from apps.integrations.ranex.exceptions import (
    RanexAuthenticationError,
    RanexConfigurationError,
    RanexInvalidResponseError,
    RanexNotFoundError,
    RanexPaymentRequiredError,
    RanexRateLimitError,
    RanexRequestError,
    RanexTimeoutError,
    RanexValidationError,
)


logger = logging.getLogger(__name__)


class RanexClient:
    def __init__(self):
        self.api_key = getattr(
            settings,
            "RUNEX_API_KEY",
            "",
        ).strip()

        self.base_url = getattr(
            settings,
            "RUNEX_BASE_URL",
            "https://kyt-api.ranex.kg",
        ).rstrip("/")

        self.timeout = getattr(
            settings,
            "RUNEX_TIMEOUT",
            30,
        )

        if not self.api_key:
            raise RanexConfigurationError(
                "RUNEX_API_KEY is not configured."
            )

        if not self.base_url:
            raise RanexConfigurationError(
                "RUNEX_BASE_URL is not configured."
            )

    def _headers(self):
        return {
            "Accept": "application/json",
            "X-API-Key": self.api_key,
        }

    def _request(
        self,
        *,
        method,
        endpoint,
        params=None,
        expect_json=True,
    ):
        url = f"{self.base_url}{endpoint}"

        try:
            response = requests.request(
                method=method,
                url=url,
                headers=self._headers(),
                params=params or None,
                timeout=self.timeout,
            )

        except requests.Timeout as exc:
            logger.exception(
                "Ranex request timed out. method=%s endpoint=%s",
                method,
                endpoint,
            )

            raise RanexTimeoutError(
                "Ranex request timed out."
            ) from exc

        except requests.RequestException as exc:
            logger.exception(
                "Ranex network request failed. "
                "method=%s endpoint=%s",
                method,
                endpoint,
            )

            raise RanexRequestError(
                f"Ranex request failed: {exc}"
            ) from exc

        if response.status_code == 401:
            raise RanexAuthenticationError(
                "Ranex authentication failed."
            )

        if response.status_code == 402:
            raise RanexPaymentRequiredError(
                "Ranex payment/subscription limit reached."
            )

        if response.status_code == 404:
            raise RanexNotFoundError(
                "Ranex resource not found."
            )

        if response.status_code == 429:
            raise RanexRateLimitError(
                "Ranex rate limit exceeded."
            )

        if response.status_code >= 400:
            body = response.text[:2000]

            logger.error(
                "Ranex HTTP error. "
                "status=%s endpoint=%s body=%s",
                response.status_code,
                endpoint,
                body,
            )

            raise RanexRequestError(
                f"Ranex HTTP {response.status_code}: "
                f"{body}"
            )

        if not expect_json:
            return response.content

        try:
            data = response.json()
        except (ValueError, json.JSONDecodeError) as exc:
            raise RanexInvalidResponseError(
                "Ranex returned invalid JSON."
            ) from exc

        if not isinstance(data, dict):
            raise RanexInvalidResponseError(
                "Ranex returned unexpected response format."
            )

        return data

    def get_usage(self):
        return self._request(
            method="GET",
            endpoint=RanexEndpoints.USAGE,
        )

    def get_risk_categories(self):
        return self._request(
            method="GET",
            endpoint=RanexEndpoints.RISK_CATEGORIES,
        )

    def get_check_history(
        self,
        *,
        from_date=None,
        to_date=None,
        page=1,
        limit=50,
    ):
        params = {
            "page": page,
            "limit": limit,
        }

        if from_date:
            params["fromDate"] = from_date

        if to_date:
            params["toDate"] = to_date

        return self._request(
            method="GET",
            endpoint=RanexEndpoints.CHECK_HISTORY,
            params=params,
        )

    def screen_address(
        self,
        *,
        network,
        address,
    ):
        network = (
            network
            or ""
        ).strip().lower()

        address = (
            address
            or ""
        ).strip()

        if network not in RanexNetwork.SUPPORTED:
            raise RanexValidationError(
                f"Unsupported Ranex network: {network}"
            )

        if not address:
            raise RanexValidationError(
                "Crypto address is required."
            )

        endpoint = RanexEndpoints.SCREEN_ADDRESS.format(
            network=quote(
                network,
                safe="",
            ),
            address=quote(
                address,
                safe="",
            ),
        )

        return self._request(
            method="GET",
            endpoint=endpoint,
        )

    def download_pdf_report(
        self,
        *,
        screening_id,
    ):
        if not screening_id:
            raise RanexValidationError(
                "screening_id is required."
            )

        endpoint = RanexEndpoints.PDF_REPORT.format(
            screening_id=screening_id,
        )

        return self._request(
            method="GET",
            endpoint=endpoint,
            expect_json=False,
        )

from __future__ import annotations

import logging
from typing import Any
from urllib.parse import quote, urljoin, urlparse

import requests
from django.conf import settings

from .exceptions import (
    RanexComplianceConfigurationError,
    RanexComplianceConnectionError,
    RanexComplianceResponseError,
)


logger = logging.getLogger(__name__)


class RanexComplianceClient:
    def __init__(
        self,
        *,
        base_url: str | None = None,
        api_key: str | None = None,
        api_secret: str | None = None,
        session: requests.Session | None = None,
    ) -> None:
        self.base_url = (
            base_url or settings.RUNEX_KYC_BASE_URL
        ).rstrip("/") + "/"
        self.api_key = api_key or settings.RUNEX_KYC_API_KEY
        self.api_secret = api_secret or settings.RUNEX_KYC_API_SECRET
        self.timeout = (
            settings.RUNEX_KYC_CONNECT_TIMEOUT,
            settings.RUNEX_KYC_READ_TIMEOUT,
        )
        self.session = session or requests.Session()
        self._validate_configuration()

    def _validate_configuration(self) -> None:
        parsed = urlparse(self.base_url)
        if parsed.scheme != "https" or not parsed.netloc:
            raise RanexComplianceConfigurationError(
                "RANEX_COMPLIANCE_BASE_URL должен быть абсолютным HTTPS URL."
            )
        if not self.api_key or not self.api_secret:
            raise RanexComplianceConfigurationError(
                "Не настроены ключи Ranex Compliance."
            )

    def get_status(self, external_user_id: str) -> dict[str, Any]:
        external_id = quote(external_user_id, safe="")
        return self.get(f"v1/applicants/{external_id}/status")

    def create_applicant(self, payload: dict[str, Any]) -> dict[str, Any]:
        return self.post("v1/applicants", data=payload)

    def get_applicant(self, external_user_id: str) -> dict[str, Any]:
        external_id = quote(external_user_id, safe="")
        return self.get(f"v1/applicants/{external_id}")

    def create_verification(
        self,
        external_user_id: str,
        *,
        level_name: str,
    ) -> dict[str, Any]:
        external_id = quote(external_user_id, safe="")
        return self._request(
            "POST",
            f"v1/applicants/{external_id}/verification-sessions",
            params={
                "verificationLevelName": level_name,
                "lang": "ru",
                "theme": "system",
                "returnUrl": settings.RUNEX_KYC_RETURN_URL,
            },
        )

    def get(self, path: str) -> dict[str, Any]:
        return self._request("GET", path)

    def post(self, path: str, *, data: dict[str, Any]) -> dict[str, Any]:
        return self._request("POST", path, data=data)

    def _request(
        self,
        method: str,
        path: str,
        *,
        data: dict[str, Any] | None = None,
        params: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        url = self._build_url(path)
        logger.info("Ranex Compliance request method=%s path=%s", method, path)

        try:
            headers = {
                "X-Api-Key": self.api_key,
                "X-Api-Secret": self.api_secret,
                "Accept": "application/json",
                "Content-Type": "application/json",
            }

            response = self.session.request(
                method,
                url,
                json=data,
                params=params,
                headers=headers,
                timeout=self.timeout,
                allow_redirects=False,
            )
        except requests.Timeout as exc:
            raise RanexComplianceConnectionError(
                "Ranex Compliance не ответил вовремя.",
                retryable=True,
            ) from exc
        except requests.RequestException as exc:
            raise RanexComplianceConnectionError(
                "Не удалось подключиться к Ranex Compliance.",
                retryable=True,
            ) from exc

        request_id = (
            response.headers.get("X-Request-ID")
            or response.headers.get("X-Correlation-ID")
        )
        try:
            body = response.json() if response.content else {}
        except ValueError as exc:
            logger.warning(
                "Ranex Compliance invalid JSON path=%s status=%s request_id=%s",
                path,
                response.status_code,
                request_id,
            )
            raise RanexComplianceResponseError(
                "Ranex Compliance вернул некорректный ответ.",
                status_code=response.status_code,
                retryable=response.status_code >= 500,
            ) from exc

        if not 200 <= response.status_code < 300:
            message = self._extract_error_message(body)
            logger.warning(
                "Ranex Compliance error path=%s status=%s request_id=%s",
                path,
                response.status_code,
                request_id,
            )
            raise RanexComplianceResponseError(
                message,
                status_code=response.status_code,
                response_data=body,
                retryable=response.status_code == 429 or response.status_code >= 500,
            )

        if not isinstance(body, dict):
            raise RanexComplianceResponseError(
                "Ranex Compliance вернул неожиданный формат ответа.",
                status_code=response.status_code,
            )

        logger.info(
            "Ranex Compliance response path=%s status=%s request_id=%s",
            path,
            response.status_code,
            request_id,
        )
        return body

    def _build_url(self, path: str) -> str:
        url = urljoin(self.base_url, path.lstrip("/"))
        if urlparse(url).netloc != urlparse(self.base_url).netloc:
            raise RanexComplianceConfigurationError("Некорректный endpoint Ranex.")
        return url

    @staticmethod
    def _extract_error_message(body: Any) -> str:
        if isinstance(body, dict):
            for key in ("message", "detail", "error", "title"):
                value = body.get(key)
                if isinstance(value, str) and value.strip():
                    return value.strip()
            errors = body.get("errors")
            if isinstance(errors, list) and errors:
                first = errors[0]
                if isinstance(first, dict) and first.get("message"):
                    return str(first["message"])
        return "Ranex Compliance отклонил запрос."

import logging
from django.conf import settings
from django.db import transaction
from drf_spectacular.utils import extend_schema
from rest_framework import serializers, status
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response

from rest_framework.views import APIView

from apps.integrations.ranex_compliance.client import RanexComplianceClient
from apps.integrations.ranex_compliance.exceptions import RanexComplianceResponseError
from .models import KYCVerification
from .serializers import KYCSerializer, KYCStartResponseSerializer

logger = logging.getLogger(__name__)


class KYBStartSerializer(serializers.Serializer):
    company_name = serializers.CharField(max_length=255)
    registration_number = serializers.CharField(max_length=128)
    legal_form = serializers.CharField(max_length=128, required=False)
    incorporation_date = serializers.DateField(required=False)
    registered_address = serializers.CharField(max_length=512, required=False)
    country = serializers.CharField(max_length=100, required=False)
    phone = serializers.CharField(max_length=32, required=False)


class KYBStartThrottle(UserRateThrottle):
    rate = "3/hour"


class KYBStartView(APIView):
    permission_classes = [IsAuthenticated]
    throttle_classes = [KYBStartThrottle]

    @extend_schema(
        tags=["KYC Verification"],
        summary="Начать KYB юридического лица через Ranex",
        request=KYBStartSerializer,
        responses={200: KYCStartResponseSerializer},
    )
    def post(self, request):
        user = request.user
        if not user.is_active or not user.is_email_verified:
            return Response({"detail": "Подтвердите email"}, status=403)
        if user.account_type != "legal":
            return Response({"detail": "KYB доступен только юридическим лицам"}, status=400)
        serializer = KYBStartSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        existing = KYCVerification.objects.filter(user=user).first()
        if existing and existing.status in ("approved", "pending"):
            return Response({"detail": "Верификация уже начата или завершена"}, status=409)

        data = serializer.validated_data
        level = settings.RUNEX_CATEGORY_KYB_LEVEL
        applicant_payload = {
            "externalUserId": str(user.pk),
            "type": "company",
            "verificationLevelName": level,
            "companyName": data["company_name"],
            "taxId": data["registration_number"],
            "email": user.email,
        }
        for source, target in (
            ("legal_form", "legalForm"),
            ("registered_address", "registeredAddress"),
            ("country", "country"),
            ("phone", "phone"),
        ):
            if data.get(source):
                applicant_payload[target] = data[source]
        if data.get("incorporation_date"):
            applicant_payload["incorporationDate"] = data["incorporation_date"].isoformat()

        # No network calls under a database lock.
        try:
            client = RanexComplianceClient()
            try:
                applicant = client.create_applicant(applicant_payload)
            except RanexComplianceResponseError as exc:
                if exc.status_code != 409:
                    raise
                applicant = client.get_applicant(str(user.pk))
            verification = client.create_verification(str(user.pk), level_name=level)
            sdk_link = verification.get("sdkLink") or {}
            if not isinstance(sdk_link, dict):
                sdk_link = {}
            url = (verification.get("url") or verification.get("verificationUrl")
                   or sdk_link.get("url"))
            applicant_id = applicant.get("id") or applicant.get("applicantId")
            if not isinstance(url, str) or not url.startswith("https://") or not applicant_id:
                raise ValueError("Ranex не вернул ссылку или applicant ID")
        except Exception:
            logger.exception("KYB start failed for user %s", user.pk)
            return Response({"detail": "Не удалось запустить KYB"}, status=502)

        with transaction.atomic():
            # Serialize writes for this user and do not override a completed check.
            type(user).objects.select_for_update().get(pk=user.pk)
            latest = KYCVerification.objects.filter(user=user).first()
            if latest and latest.status in ("approved", "pending"):
                return Response({"detail": "Статус проверки изменился, обновите страницу"}, status=409)
            record = KYCVerification.objects.create(
                user=user, level_name=level, status=KYCVerification.Status.PENDING,
                external_applicant_id=str(applicant_id)[:128],
                external_check_id=str(verification.get("id") or verification.get("sessionId") or "")[:128],
            )
        return Response({"verification": KYCSerializer(record).data, "verification_url": url})

from django.conf import settings
from django.utils import timezone
from apps.kyc.models import KYCVerification
from apps.integrations.ranex_compliance.client import RanexComplianceClient


def start_kyc(*, user, applicant_data):
    if not user.is_active:
        raise ValueError('Неактивный пользователь')
    level = settings.RUNEX_CATEGORY_KYC_LEVEL
    record = KYCVerification.objects.create(user=user, level_name=level)
    client = RanexComplianceClient()
    try:
        payload = dict(applicant_data)
        payload['externalUserId'] = str(user.pk)
        applicant = client.create_applicant(payload)
        session = client.create_verification(str(user.pk), level_name=level)
        record.external_applicant_id = str(applicant.get('id') or applicant.get('applicantId') or '')[:128]
        record.external_check_id = str(session.get('id') or session.get('sessionId') or '')[:128]
        record.status = KYCVerification.Status.PENDING
        record.save(update_fields=['external_applicant_id','external_check_id','status','updated_at'])
        return record, session
    except Exception:
        record.status = KYCVerification.Status.ERROR
        record.save(update_fields=['status','updated_at'])
        raise


def refresh_kyc_status(*, record):
    # Never overwrite a terminal webhook decision with a stale polling result.
    record.refresh_from_db()
    if record.status in (KYCVerification.Status.APPROVED, KYCVerification.Status.REJECTED):
        return record
    response = RanexComplianceClient().get_status(str(record.user_id))
    status = str(response.get('reviewStatus') or response.get('status') or '').lower()
    answer = str(response.get('reviewAnswer') or '').upper()
    if status in {'completed', 'approved'} and answer in {'GREEN', 'APPROVED'}:
        record.status = KYCVerification.Status.APPROVED
        record.verified_at = timezone.now()
    elif status in {'completed', 'rejected'} and answer in {'RED', 'REJECTED'}:
        record.status = KYCVerification.Status.REJECTED
    else:
        record.status = KYCVerification.Status.PENDING
    record.save(update_fields=['status','verified_at','updated_at'])
    return record

import base64
import hashlib
import hmac
import time
from unittest.mock import patch

from django.test import SimpleTestCase, override_settings

from .webhook import SignatureError, verify_signature


class RanexWebhookSignatureTests(SimpleTestCase):
    @override_settings(RUNEX_KYC_WEBHOOK_SECRET=base64.b64encode(b"a" * 32).decode())
    def test_valid_and_tampered(self):
        body = b'{"type":"applicantPending"}'
        stamp = str(int(time.time() * 1000))
        digest = hmac.new(b"a" * 32, stamp.encode() + b"." + body, hashlib.sha256).hexdigest()
        header = f"t={stamp},v1={digest}"
        verify_signature(body, header)
        with self.assertRaises(SignatureError):
            verify_signature(body + b" ", header)

    @override_settings(RUNEX_KYC_WEBHOOK_SECRET=base64.b64encode(b"a" * 32).decode())
    def test_expired(self):
        stamp = str(int(time.time() * 1000) - 600_000)
        digest = hmac.new(b"a" * 32, stamp.encode() + b"." + b"{}", hashlib.sha256).hexdigest()
        with self.assertRaises(SignatureError):
            verify_signature(b"{}", f"t={stamp},v1={digest}")

import logging

from django.conf import settings
from drf_spectacular.utils import extend_schema
from rest_framework import status
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response
from rest_framework.throttling import UserRateThrottle
from rest_framework.views import APIView

from .models import KYCVerification
from .serializers import (
    KYCSerializer,
    KYCMeSerializer,
    KYCStartSerializer,
    KYCStartResponseSerializer,
)
from .services import start_kyc, refresh_kyc_status

logger = logging.getLogger(__name__)


class KYCStartThrottle(UserRateThrottle):
    rate = "3/hour"


class KYCMeView(APIView):
    permission_classes = [IsAuthenticated]

    @extend_schema(
        tags=["KYC Verification"],
        summary="Получить статус верификации",
        responses={200: KYCMeSerializer},
    )
    def get(self, request):
        record = (
            KYCVerification.objects
            .filter(user=request.user)
            .first()
        )

        if record is None:
            return Response({
                "status": "not_started",
                "verification": None,
            })

        return Response({
            "status": record.status,
            "verification": KYCSerializer(record).data,
        })


class KYCStartView(APIView):
    permission_classes = [IsAuthenticated]
    throttle_classes = [KYCStartThrottle]

    @extend_schema(
        tags=["KYC Verification"],
        summary="Начать верификацию через Ranex",
        request=KYCStartSerializer,
        responses={200: KYCStartResponseSerializer},
    )
    def post(self, request):
        user = request.user

        if not user.is_email_verified:
            return Response(
                {"detail": "Сначала подтвердите email"},
                status=status.HTTP_403_FORBIDDEN,
            )

        if user.account_type != "individual":
            return Response(
                {"detail": "Для юридических лиц требуется отдельный процесс KYC"},
                status=status.HTTP_400_BAD_REQUEST,
            )

        existing = (
            KYCVerification.objects
            .filter(user=user)
            .first()
        )

        if existing and existing.status == "approved":
            return Response(
                {"detail": "Верификация уже пройдена"},
                status=status.HTTP_409_CONFLICT,
            )

        if existing and existing.status == "pending":
            return Response(
                {"detail": "Верификация уже находится на рассмотрении"},
                status=status.HTTP_409_CONFLICT,
            )

        serializer = KYCStartSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        data = serializer.validated_data

        applicant_data = {
            "email": user.email,
            "firstName": data["first_name"],
            "lastName": data["last_name"],
        }

        if data.get("phone"):
            applicant_data["phone"] = data["phone"]

        try:
            record, session = start_kyc(
                user=user,
                applicant_data=applicant_data,
            )
        except Exception:
            logger.exception(
                "KYC start failed for user %s",
                user.pk,
            )
            return Response(
                {"detail": "Не удалось запустить верификацию"},
                status=status.HTTP_502_BAD_GATEWAY,
            )

        verification_url = (
            session.get("verificationUrl")
            or session.get("applicantLink")
            or session.get("url")
        )

        if (
            not isinstance(verification_url, str)
            or not verification_url.startswith("https://")
        ):
            return Response(
                {"detail": "Ranex не вернул корректную ссылку"},
                status=status.HTTP_502_BAD_GATEWAY,
            )

        return Response({
            "verification": KYCSerializer(record).data,
            "verification_url": verification_url,
        })


class KYCRefreshView(APIView):
    permission_classes = [IsAuthenticated]
    throttle_classes = [KYCStartThrottle]

    @extend_schema(
        tags=["KYC Verification"],
        summary="Обновить статус KYC из Ranex",
        responses={200: KYCSerializer},
    )
    def post(self, request):
        record = (
            KYCVerification.objects
            .filter(user=request.user)
            .first()
        )

        if record is None:
            return Response(
                {"detail": "Верификация не начата"},
                status=status.HTTP_404_NOT_FOUND,
            )

        if record.status == "approved":
            return Response(KYCSerializer(record).data)

        try:
            record = refresh_kyc_status(record=record)
        except Exception:
            logger.exception(
                "KYC status refresh failed for %s",
                request.user.pk,
            )
            return Response(
                {"detail": "Не удалось получить статус Ranex"},
                status=status.HTTP_502_BAD_GATEWAY,
            )

        return Response(KYCSerializer(record).data)

"""Ranex Compliance webhook, adapted from WTP signature verification.

Provider contract: X-Webhook-Signature: t=<unix-ms>,v1=<hex-hmac>;
HMAC-SHA256(base64-decoded 32-byte secret, timestamp + b'.' + raw body).
"""
import base64
import hashlib
import hmac
import json
import logging
import time
from datetime import datetime, timezone as dt_timezone

from django.conf import settings
from django.db import transaction
from django.utils import timezone
from django.utils.dateparse import parse_datetime
from rest_framework.permissions import AllowAny
from rest_framework.response import Response
from rest_framework.views import APIView

from .models import KYCVerification

logger = logging.getLogger(__name__)
EVENTS = {
    "applicantReviewed", "applicantAwaitingOperatorReview", "applicantOnHold",
    "applicantWorkflowFailed", "applicantExpired", "applicantPending",
}


class SignatureError(ValueError):
    pass


class WebhookConfigurationError(RuntimeError):
    pass


def verify_signature(body: bytes, header: str) -> None:
    secret = settings.RUNEX_KYC_WEBHOOK_SECRET
    if not secret:
        raise WebhookConfigurationError("Missing Ranex webhook secret")
    try:
        parts = dict(part.strip().split("=", 1) for part in header.split(",") if "=" in part)
        stamp, digest = parts["t"], parts["v1"]
        if not stamp.isascii() or not stamp.isdigit():
            raise ValueError("Invalid timestamp")
        if len(digest) != 64 or any(c not in "0123456789abcdefABCDEF" for c in digest):
            raise ValueError("Invalid digest")
        if abs(int(time.time() * 1000) - int(stamp)) > 300_000:
            raise ValueError("Expired timestamp")
        key = base64.b64decode(secret, validate=True)
        if len(key) != 32:
            raise ValueError("Invalid key length")
    except (KeyError, ValueError, TypeError) as exc:
        raise SignatureError("Invalid signature") from exc
    expected = hmac.new(key, stamp.encode("ascii") + b"." + body, hashlib.sha256).hexdigest()
    if not hmac.compare_digest(expected, digest.lower()):
        raise SignatureError("Invalid signature")


def event_datetime(payload):
    ms = payload.get("createdAtMs")
    if isinstance(ms, bool):
        raise ValueError("Invalid event time")
    if isinstance(ms, (int, float)):
        try:
            return datetime.fromtimestamp(ms / 1000, tz=dt_timezone.utc)
        except (ValueError, OverflowError, OSError) as exc:
            raise ValueError("Invalid event time") from exc
    value = payload.get("createdAt")
    parsed = parse_datetime(value) if isinstance(value, str) else None
    if parsed is None or timezone.is_naive(parsed):
        raise ValueError("Missing or invalid event time")
    return parsed


def process_event(payload: dict, digest: str) -> dict:
    kind = payload.get("type")
    if kind not in EVENTS:
        return {"ok": True, "ignored": True}
    external_user_id = payload.get("externalUserId")
    applicant_id = payload.get("applicantId")
    if not isinstance(external_user_id, str) or not external_user_id or not isinstance(applicant_id, str) or not applicant_id:
        raise ValueError("Missing applicant identifiers")
    sandbox = payload.get("sandboxMode")
    if not isinstance(sandbox, bool):
        raise ValueError("Missing sandboxMode")
    if sandbox != str(settings.RUNEX_KYC_API_KEY).startswith("pk_test_"):
        return {"ok": True, "ignored": True}
    occurred = event_datetime(payload)
    review = payload.get("reviewResult") or {}
    if not isinstance(review, dict):
        raise ValueError("Invalid reviewResult")
    answer = review.get("reviewAnswer")
    if kind == "applicantReviewed" and answer not in ("GREEN", "RED"):
        raise ValueError("Invalid reviewAnswer")

    # Ranex test events may use non-UUID externalUserId values.
    # Never query a UUIDField with an invalid identifier.
    from django.contrib.auth import get_user_model
    from django.core.exceptions import ValidationError

    try:
        external_user_id = get_user_model()._meta.pk.to_python(external_user_id)
    except (ValidationError, ValueError, TypeError):
        logger.info("Ranex webhook ignored: invalid external user ID")
        return {"ok": True, "ignored": True, "reason": "unknown_user"}

    with transaction.atomic():
        # Match the exact applicant AND the external user; never approve by user ID alone.
        records = list(KYCVerification.objects.select_for_update().filter(
            user_id=external_user_id, external_applicant_id=applicant_id,
            provider="ranex",
        ).order_by("-created_at")[:2])
        if not records:
            # The applicant may have been created before the local start was committed.
            # Retry only if a local start is in progress; otherwise ignore foreign applicants.
            if KYCVerification.objects.filter(user_id=external_user_id, status="not_started").exists():
                raise RuntimeError("Verification start in progress")
            return {"ok": True, "ignored": True}
        record = records[0]
        level = payload.get("levelName")
        if level is not None and (not isinstance(level, str) or level.casefold() != record.level_name.casefold()):
            return {"ok": True, "ignored": True}
        previous = record.provider_payload if isinstance(record.provider_payload, dict) else {}
        if previous.get("webhook_sha256") == digest:
            return {"ok": True, "duplicate": True}
        previous_date = parse_datetime(previous.get("event_at", "")) if isinstance(previous.get("event_at"), str) else None
        if previous_date and occurred <= previous_date:
            return {"ok": True, "ignored": True, "reason": "old_event"}
        if occurred < record.created_at:
            return {"ok": True, "ignored": True, "reason": "before_start"}
        # A terminal decision is immutable through webhook; subsequent changes require manual review.
        if record.status in (KYCVerification.Status.APPROVED, KYCVerification.Status.REJECTED):
            return {"ok": True, "ignored": True, "reason": "terminal"}
        if kind == "applicantReviewed":
            record.status = KYCVerification.Status.APPROVED if answer == "GREEN" else KYCVerification.Status.REJECTED
            record.verified_at = timezone.now() if answer == "GREEN" else None
        elif kind == "applicantExpired":
            record.status = KYCVerification.Status.EXPIRED
        elif kind == "applicantWorkflowFailed":
            record.status = KYCVerification.Status.ERROR
        else:
            record.status = KYCVerification.Status.PENDING
        # Deliberately do not store document data, raw body or personal information.
        record.provider_payload = {"event_type": kind, "event_at": occurred.isoformat(),
                                   "webhook_sha256": digest, "review_answer": answer if kind == "applicantReviewed" else None}
        record.save(update_fields=["status", "verified_at", "provider_payload", "updated_at"])
    return {"ok": True, "status": record.status}


class RanexWebhookView(APIView):
    authentication_classes = []
    permission_classes = [AllowAny]
    parser_classes = []

    def post(self, request):
        body = request.body
        if len(body) > 262_144:
            return Response({"detail": "Payload too large"}, status=413)
        try:
            verify_signature(body, request.headers.get("X-Webhook-Signature", ""))
        except SignatureError:
            return Response({"detail": "Invalid signature"}, status=401)
        except WebhookConfigurationError:
            return Response({"detail": "Webhook unavailable"}, status=503)
        try:
            payload = json.loads(body)
            if not isinstance(payload, dict):
                raise ValueError("Invalid payload")
            result = process_event(payload, hashlib.sha256(body).hexdigest())
            return Response(result)
        except (ValueError, UnicodeDecodeError):
            return Response({"detail": "Invalid payload"}, status=400)
        except Exception as exc:
            logger.error("Ranex webhook processing failed: %s", type(exc).__name__)
            return Response({"detail": "Temporary processing error"}, status=503)

from django.contrib.auth.base_user import BaseUserManager
class UserManager(BaseUserManager):
    def create_user(self, email, password=None, **extra_fields):
        if not email:
            raise ValueError('Email обязателен')
        email = self.normalize_email(email).lower()
        user = self.model(email=email, **extra_fields)
        user.set_password(password)
        user.save(using=self._db)
        return user
    def create_superuser(self, email, password=None, **extra_fields):
        extra_fields.setdefault('is_staff', True)
        extra_fields.setdefault('is_superuser', True)
        extra_fields.setdefault('is_active', True)
        extra_fields.setdefault('is_email_verified', True)
        if not extra_fields['is_staff'] or not extra_fields['is_superuser']:
            raise ValueError('Superuser must have is_staff=True and is_superuser=True')
        return self.create_user(email, password, **extra_fields)

def password_fingerprint(user):
    return hmac.new(
        settings.SECRET_KEY.encode(),
        user.password.encode(),
        hashlib.sha256,
    ).hexdigest()


def invalid_code():
    return Response(
        {"detail": "Неверный или истёкший код"},
        status=400,
    )


class PasswordResetRequestView(APIView):
    permission_classes = [AllowAny]
    authentication_classes = []
    throttle_classes = [PasswordResetThrottle]

    @extend_schema(
        tags=["Password Reset"],
        request=PasswordResetRequestSerializer,
        responses={200: DetailSerializer},
    )
    def post(self, request):
        serializer = PasswordResetRequestSerializer(
            data=request.data
        )
        serializer.is_valid(raise_exception=True)

        email = serializer.validated_data["email"].strip().lower()

        response = {
            "detail": (
                "Если email зарегистрирован, "
                "код восстановления будет отправлен"
            )
        }

        user = User.objects.filter(
            email__iexact=email,
            is_active=True,
        ).first()

        if user is None:
            return Response(response)

        now = timezone.now()

        latest = EmailVerificationCode.objects.filter(
            user=user,
            code_hash__startswith=RESET_PREFIX,
        ).first()

        if latest and latest.created_at > now - timedelta(seconds=60):
            return Response(response)

        code = f"{secrets.randbelow(1_000_000):06d}"

        record = EmailVerificationCode.objects.create(
            user=user,
            code_hash=RESET_PREFIX + make_password(code),
            expires_at=now + timedelta(minutes=10),
        )

        try:
            send_mail(
                subject="BNBroker — восстановление пароля",
                message=(
                    f"Ваш код: {code}\n"
                    "Код действителен 10 минут.\n"
                    "Если вы не запрашивали сброс, "
                    "проигнорируйте письмо."
                ),
                from_email=settings.DEFAULT_FROM_EMAIL,
                recipient_list=[user.email],
                fail_silently=False,
            )
        except Exception:
            record.delete()
            logger.exception("Password reset email delivery failed")
            return Response(response)

        EmailVerificationCode.objects.filter(
            user=user,
            code_hash__startswith=RESET_PREFIX,
            used_at__isnull=True,
        ).exclude(pk=record.pk).update(used_at=timezone.now())

        return Response(response)


class PasswordResetVerifyView(APIView):
    permission_classes = [AllowAny]
    authentication_classes = []
    throttle_classes = [PasswordResetThrottle]

    @extend_schema(
        tags=["Password Reset"],
        request=PasswordResetVerifySerializer,
        responses={
            200: ResetTokenSerializer,
            400: DetailSerializer,
        },
    )
    def post(self, request):
        serializer = PasswordResetVerifySerializer(
            data=request.data
        )
        serializer.is_valid(raise_exception=True)

        email = serializer.validated_data["email"].strip().lower()
        code = serializer.validated_data["code"]

        with transaction.atomic():
            user = User.objects.filter(
                email__iexact=email,
                is_active=True,
            ).first()

            if not user:
                return invalid_code()

            record = (
                EmailVerificationCode.objects
                .select_for_update()
                .filter(
                    user=user,
                    code_hash__startswith=RESET_PREFIX,
                )
                .first()
            )

            now = timezone.now()

            if (
                not record
                or record.used_at is not None
                or record.expires_at <= now
                or record.attempts >= 5
            ):
                return invalid_code()

            record.attempts += 1
            record.save(update_fields=["attempts"])

            if not check_password(
                code,
                record.code_hash[len(RESET_PREFIX):],
            ):
                return invalid_code()

            record.used_at = now
            record.save(update_fields=["used_at"])

            token = signing.dumps(
                {
                    "uid": str(user.pk),
                    "rid": record.pk,
                    "pwd": password_fingerprint(user),
                },
                salt=RESET_SALT,
            )

        return Response({
            "reset_token": token,
            "expires_in": 300,
        })


class PasswordResetConfirmView(APIView):
    permission_classes = [AllowAny]
    authentication_classes = []
    throttle_classes = [PasswordResetThrottle]

    @extend_schema(
        tags=["Password Reset"],
        request=PasswordResetConfirmSerializer,
        responses={
            200: DetailSerializer,
            400: DetailSerializer,
        },
    )
    def post(self, request):
        serializer = PasswordResetConfirmSerializer(
            data=request.data
        )
        serializer.is_valid(raise_exception=True)

        data = serializer.validated_data

        try:
            payload = signing.loads(
                data["reset_token"],
                salt=RESET_SALT,
                max_age=300,
            )
        except (signing.BadSignature, signing.SignatureExpired):
            return Response(
                {"detail": "Токен недействителен или истёк"},
                status=400,
            )

        with transaction.atomic():
            user = (
                User.objects
                .select_for_update()
                .filter(
                    pk=payload.get("uid"),
                    is_active=True,
                )
                .first()
            )

            if user is None:
                return Response(
                    {"detail": "Токен недействителен"},
                    status=400,
                )

            record = (
                EmailVerificationCode.objects
                .select_for_update()
                .filter(
                    pk=payload.get("rid"),
                    user=user,
                    code_hash__startswith=RESET_PREFIX,
                    used_at__isnull=False,
                )
                .first()
            )

            if (
                record is None
                or record.attempts > 5
                or not hmac.compare_digest(
                    str(payload.get("pwd", "")),
                    password_fingerprint(user),
                )
            ):
                return Response(
                    {"detail": "Токен недействителен"},
                    status=400,
                )

            validate_password(data["new_password"], user=user)

            user.set_password(data["new_password"])
            user.save(update_fields=["password"])

        return Response({
            "detail": "Пароль успешно изменён"
        })

from rest_framework.exceptions import AuthenticationFailed

from rest_framework_simplejwt.authentication import JWTAuthentication

from .models import UserTwoFactor


def validate_security_claims(user, token):
    config = UserTwoFactor.objects.filter(
        user=user
    ).only(
        "enabled",
        "auth_version",
    ).first()

    current_version = config.auth_version if config else 0
    token_version = token.get("auth_version", 0)

    if (
        type(token_version) is not int
        or token_version != current_version
    ):
        raise AuthenticationFailed(
            "Сессия устарела. Войдите заново.",
            code="session_revoked",
        )

    if config and config.enabled:
        if token.get("mfa_verified") is not True:
            raise AuthenticationFailed(
                "Требуется двухфакторная аутентификация.",
                code="mfa_required",
            )


class TwoFactorJWTAuthentication(JWTAuthentication):
    def get_user(self, validated_token):
        user = super().get_user(validated_token)

        validate_security_claims(
            user,
            validated_token,
        )

        return user

from cryptography.fernet import Fernet, InvalidToken
from django.conf import settings
from django.core.exceptions import ImproperlyConfigured


def get_cipher():
    key = getattr(settings, "TOTP_ENCRYPTION_KEY", "")

    if not key:
        raise ImproperlyConfigured(
            "TOTP_ENCRYPTION_KEY is not configured"
        )

    try:
        return Fernet(key.encode("ascii"))
    except (ValueError, TypeError) as exc:
        raise ImproperlyConfigured(
            "Invalid TOTP_ENCRYPTION_KEY"
        ) from exc


def encrypt_totp_secret(secret: str) -> str:
    if not secret:
        raise ValueError("Empty TOTP secret")

    return get_cipher().encrypt(
        secret.encode("utf-8")
    ).decode("ascii")


def decrypt_totp_secret(encrypted_secret: str) -> str:
    if not encrypted_secret:
        raise ValueError("Empty encrypted TOTP secret")

    try:
        return get_cipher().decrypt(
            encrypted_secret.encode("ascii")
        ).decode("utf-8")
    except InvalidToken as exc:
        raise ValueError(
            "Unable to decrypt TOTP secret"
        ) from exc

def issue_tokens(user, mfa_verified=False):
    config = UserTwoFactor.objects.filter(user=user).first()

    refresh = RefreshToken.for_user(user)

    refresh["auth_version"] = (
        config.auth_version if config else 0
    )
    refresh["mfa_verified"] = bool(mfa_verified)

    return {
        "refresh": str(refresh),
        "access": str(refresh.access_token),
    }


def record_login(user, request):
    UserLoginHistory.objects.create(
        user=user,
        ip_address=request.META.get("REMOTE_ADDR"),
        user_agent=request.META.get(
            "HTTP_USER_AGENT", ""
        )[:512],
        successful=True,
    )


class SecureLoginView(APIView):
    permission_classes = [permissions.AllowAny]
    throttle_classes = [LoginThrottle]

    @extend_schema(
        tags=["Authentication / 2FA"],
        summary="Вход с поддержкой Google Authenticator",
        request=SecureLoginSerializer,
        responses={200: None},
    )
    def post(self, request):
        serializer = SecureLoginSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        user = authenticate(
            request=request,
            username=serializer.validated_data["email"],
            password=serializer.validated_data["password"],
        )

        if not user or not user.is_active or not user.is_email_verified:
            return Response(
                {"detail": "Неверный email или пароль"},
                status=status.HTTP_401_UNAUTHORIZED,
            )

        config = UserTwoFactor.objects.filter(user=user).first()

        if config and config.enabled:
            raw_token = secrets.token_urlsafe(32)
            token_hash = hashlib.sha256(
                raw_token.encode()
            ).hexdigest()

            TwoFactorLoginChallenge.objects.create(
                user=user,
                token_hash=token_hash,
                expires_at=timezone.now() + timedelta(minutes=5),
            )

            return Response({
                "requires_2fa": True,
                "challenge_token": raw_token,
                "expires_in": 300,
            })

        tokens = issue_tokens(user, mfa_verified=False)
        record_login(user, request)

        return Response({
            "requires_2fa": False,
            **tokens,
        })


class SecureLoginVerifyView(APIView):
    permission_classes = [permissions.AllowAny]
    throttle_classes = [LoginThrottle]

    @extend_schema(
        tags=["Authentication / 2FA"],
        summary="Подтвердить код Google Authenticator при входе",
        request=VerifyLoginSerializer,
        responses={200: None},
    )
    @transaction.atomic
    def post(self, request):
        serializer = VerifyLoginSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        raw_token = serializer.validated_data["challenge_token"]
        token_hash = hashlib.sha256(
            raw_token.encode()
        ).hexdigest()

        challenge = (
            TwoFactorLoginChallenge.objects
            .select_for_update()
            .select_related("user")
            .filter(token_hash=token_hash)
            .first()
        )

        if (
            not challenge
            or challenge.used_at is not None
            or challenge.expires_at <= timezone.now()
            or challenge.attempts >= 5
        ):
            return Response(
                {"detail": "Запрос на вход недействителен"},
                status=status.HTTP_400_BAD_REQUEST,
            )

        challenge.attempts += 1
        challenge.save(update_fields=["attempts"])

        try:
            valid = verify_totp(
                challenge.user,
                serializer.validated_data["code"],
            )
        except TwoFactorError:
            valid = False

        if not valid:
            return Response(
                {"detail": "Неверный код Google Authenticator"},
                status=status.HTTP_400_BAD_REQUEST,
            )

        challenge.used_at = timezone.now()
        challenge.save(update_fields=["used_at"])

        tokens = issue_tokens(
            challenge.user,
            mfa_verified=True,
        )

        record_login(challenge.user, request)

        return Response({
            "requires_2fa": False,
            **tokens,
        })

class SecureTokenRefreshSerializer(TokenRefreshSerializer):

    def validate(self, attrs):
        try:
            token = RefreshToken(attrs["refresh"])
            user_id = token.get("user_id")

            if not user_id:
                raise serializers.ValidationError(
                    "Недействительный refresh-токен"
                )

            user = User.objects.filter(
                pk=user_id,
                is_active=True,
                is_email_verified=True,
            ).first()

            if not user:
                raise serializers.ValidationError(
                    "Пользователь недоступен"
                )

            validate_security_claims(user, token)

        except Exception as exc:
            from rest_framework.exceptions import AuthenticationFailed
            from rest_framework_simplejwt.exceptions import TokenError

            if isinstance(
                exc,
                (AuthenticationFailed, TokenError, ValueError, TypeError),
            ):
                raise serializers.ValidationError(
                    "Сессия недействительна. Войдите заново."
                ) from exc

            raise

        return super().validate(attrs)


class SecureTokenRefreshView(TokenRefreshView):
    serializer_class = SecureTokenRefreshSerializer

import time
from datetime import timedelta

import pyotp

from django.db import transaction
from django.utils import timezone

from .models import UserTwoFactor
from .two_factor_crypto import decrypt_totp_secret


class TwoFactorError(Exception):
    pass


def verify_totp(user, code):
    if not isinstance(code, str) or not (
        len(code) == 6 and code.isascii() and code.isdigit()
    ):
        raise TwoFactorError("Неверный формат кода")

    with transaction.atomic():
        config = (
            UserTwoFactor.objects
            .select_for_update()
            .filter(user=user)
            .first()
        )

        if not config or not config.enabled:
            raise TwoFactorError("2FA не подключена")

        now = timezone.now()

        if config.locked_until and config.locked_until > now:
            raise TwoFactorError(
                "Слишком много попыток. Повторите позже."
            )

        secret = decrypt_totp_secret(
            config.encrypted_secret
        )

        totp = pyotp.TOTP(secret)
        current_step = int(time.time()) // totp.interval

        matched_step = None

        for step in (
            current_step - 1,
            current_step,
            current_step + 1,
        ):
            if step <= config.last_used_step:
                continue

            if pyotp.utils.strings_equal(
                totp.at(step * totp.interval),
                code,
            ):
                matched_step = step
                break

        if matched_step is None:
            config.failed_attempts += 1

            if config.failed_attempts >= 5:
                config.locked_until = (
                    now + timedelta(minutes=15)
                )
                config.failed_attempts = 0

            config.save(update_fields=[
                "failed_attempts",
                "locked_until",
            ])

            return False

        config.last_used_step = matched_step
        config.failed_attempts = 0
        config.locked_until = None

        config.save(update_fields=[
            "last_used_step",
            "failed_attempts",
            "locked_until",
        ])

        return True

class TwoFactorSetupView(APIView):
    permission_classes = [permissions.IsAuthenticated]
    throttle_classes = [TwoFactorThrottle]

    @extend_schema(
        tags=["Security / 2FA"],
        summary="Получить QR-код Google Authenticator",
        request=TwoFactorSetupSerializer,
        responses={200: TwoFactorSetupResponseSerializer},
    )
    @transaction.atomic
    def post(self, request):
        serializer = TwoFactorSetupSerializer(
            data=request.data
        )
        serializer.is_valid(raise_exception=True)

        if not request.user.check_password(
            serializer.validated_data["current_password"]
        ):
            return Response(
                {"detail": "Неверный пароль"},
                status=status.HTTP_400_BAD_REQUEST,
            )

        obj, _ = UserTwoFactor.objects.select_for_update().get_or_create(
            user=request.user
        )

        if obj.enabled:
            return Response(
                {"detail": "2FA уже подключена"},
                status=status.HTTP_409_CONFLICT,
            )

        secret = pyotp.random_base32()

        obj.encrypted_pending_secret = encrypt_totp_secret(secret)
        obj.pending_created_at = timezone.now()
        obj.save(update_fields=[
            "encrypted_pending_secret",
            "pending_created_at",
        ])

        totp = pyotp.TOTP(secret)

        uri = totp.provisioning_uri(
            name=request.user.email,
            issuer_name="BNBroker",
        )

        image = qrcode.make(uri)

        buffer = io.BytesIO()
        image.save(buffer, format="PNG")

        qr_base64 = base64.b64encode(
            buffer.getvalue()
        ).decode("ascii")

        response = Response({
            "secret": secret,
            "otpauth_url": uri,
            "qr_code": "data:image/png;base64," + qr_base64,
        })

        response["Cache-Control"] = "no-store"
        response["Pragma"] = "no-cache"

        return response

class TwoFactorConfirmView(APIView):
    permission_classes = [permissions.IsAuthenticated]
    throttle_classes = [TwoFactorThrottle]

    @extend_schema(
        tags=["Security / 2FA"],
        summary="Подтвердить Google Authenticator",
        request=TwoFactorConfirmSerializer,
        responses={200: None},
    )
    @transaction.atomic
    def post(self, request):
        serializer = TwoFactorConfirmSerializer(
            data=request.data
        )
        serializer.is_valid(raise_exception=True)

        obj = UserTwoFactor.objects.select_for_update().filter(
            user=request.user
        ).first()

        if not obj or not obj.encrypted_pending_secret:
            return Response(
                {"detail": "Сначала выполните настройку 2FA"},
                status=status.HTTP_400_BAD_REQUEST,
            )

        if obj.enabled:
            return Response(
                {"detail": "2FA уже подключена"},
                status=status.HTTP_409_CONFLICT,
            )

        if (
            not obj.pending_created_at
            or obj.pending_created_at
            < timezone.now() - timedelta(minutes=10)
        ):
            return Response(
                {"detail": "Время настройки истекло"},
                status=status.HTTP_400_BAD_REQUEST,
            )

        from .two_factor_crypto import decrypt_totp_secret

        secret = decrypt_totp_secret(
            obj.encrypted_pending_secret
        )

        code = serializer.validated_data["code"]

        if not pyotp.TOTP(secret).verify(
            code,
            valid_window=1,
        ):
            return Response(
                {"detail": "Неверный код Google Authenticator"},
                status=status.HTTP_400_BAD_REQUEST,
            )

        import time
        from .two_factor_login import issue_tokens

        totp = pyotp.TOTP(secret)
        current_step = int(time.time()) // totp.interval

        matched_step = next(
            (
                step
                for step in (
                    current_step - 1,
                    current_step,
                    current_step + 1,
                )
                if pyotp.utils.strings_equal(
                    totp.at(step * totp.interval),
                    code,
                )
            ),
            None,
        )

        if matched_step is None:
            return Response(
                {"detail": "Код устарел. Введите новый код."},
                status=status.HTTP_400_BAD_REQUEST,
            )

        obj.encrypted_secret = obj.encrypted_pending_secret
        obj.encrypted_pending_secret = ""
        obj.pending_created_at = None
        obj.enabled = True
        obj.confirmed_at = timezone.now()
        obj.last_used_step = matched_step
        obj.failed_attempts = 0
        obj.locked_until = None
        obj.auth_version += 1

        obj.save(update_fields=[
            "encrypted_secret",
            "encrypted_pending_secret",
            "pending_created_at",
            "enabled",
            "confirmed_at",
            "last_used_step",
            "failed_attempts",
            "locked_until",
            "auth_version",
        ])

        tokens = issue_tokens(
            request.user,
            mfa_verified=True,
        )

        response = Response({
            "verified": True,
            "enabled": True,
            "detail": "Двухфакторная аутентификация включена",
            **tokens,
        })
        response["Cache-Control"] = "no-store"
        return response

"""Restricted Unfold dashboard: reads existing data, never mutates orders."""
from datetime import timedelta

from django.db.models import Count, Sum
from django.db.models.functions import TruncDate
from django.urls import reverse
from django.utils import timezone

from apps.users.models import User
from apps.exchange.models import ExchangeOrder, ExchangeQuote, ExchangeDirection, ExchangeRate
from apps.kyc.models import KYCVerification


def dashboard_callback(request, context):
    now = timezone.now()
    user = request.user
    can_orders = user.has_perm('exchange.view_exchangeorder')
    can_rates = user.has_perm('exchange.view_exchangerate')
    can_kyc = user.has_perm('kyc.view_kycverification')
    can_users = user.has_perm('users.view_user')

    values = {
        'can_orders': can_orders,
        'can_rates': can_rates,
        'can_kyc': can_kyc,
        'can_users': can_users,
        'report_generated_at': now,
        'admin_rates_url': reverse('admin:exchange_exchangerate_changelist') if can_rates else '',
        'admin_rate_add_url': reverse('admin:exchange_exchangerate_add') if user.has_perm('exchange.add_exchangerate') else '',
        'admin_directions_url': reverse('admin:exchange_exchangedirection_changelist') if user.has_perm('exchange.view_exchangedirection') else '',
        'admin_orders_url': reverse('admin:exchange_exchangeorder_changelist') if can_orders else '',
        'admin_kyc_url': reverse('admin:kyc_kycverification_changelist') if can_kyc else '',
        'admin_users_url': reverse('admin:users_user_changelist') if can_users else '',
    }

    if can_orders:
        qs = ExchangeOrder.objects.select_related('user', 'quote', 'quote__asset_network')
        # These are outstanding unpaid orders past their quote expiry;
        # they are NOT silently marked expired by this read-only dashboard.
        values.update({
            'orders_total': qs.count(),
            'orders_today': qs.filter(created_at__date=timezone.localdate()).count(),
            'orders_waiting': qs.filter(status='awaiting_payment').count(),
            'orders_review': qs.filter(status='payment_review').count(),
            'orders_processing': qs.filter(status__in=['paid', 'processing']).count(),
            'orders_completed': qs.filter(status='completed').count(),
            'orders_overdue': qs.filter(status='awaiting_payment', quote__expires_at__lte=now).count(),
            'orders_recent': list(qs.order_by('-created_at')[:10]),
            'quotes_active': ExchangeQuote.objects.filter(expires_at__gt=now).count(),
        })

        # Completed RUB gross volume, not profit or revenue.
        total = qs.filter(status='completed').aggregate(total=Sum('quote__fiat_amount'))['total']
        values['completed_fiat_amount'] = total or 0

        start = timezone.localdate() - timedelta(days=6)
        grouped = (
            qs.filter(created_at__date__gte=start)
            .annotate(day=TruncDate('created_at'))
            .values('day')
            .annotate(total=Count('pk'))
            .order_by('day')
        )
        map_days = {row['day']: row['total'] for row in grouped}
        daily = [{'label': (start + timedelta(days=i)).strftime('%d.%m'),
                  'count': map_days.get(start + timedelta(days=i), 0)} for i in range(7)]
        highest = max([1] + [day['count'] for day in daily])
        for day in daily:
            day['percent'] = max(5, round(day['count'] * 100 / highest)) if day['count'] else 0
        values['orders_chart'] = daily

    if can_rates:
        values.update({
            'rates_total': ExchangeRate.objects.count(),
            'rates_active': ExchangeRate.objects.filter(is_active=True).count(),
            'directions_active': ExchangeDirection.objects.filter(is_active=True).count(),
            'rates_recent': list(ExchangeRate.objects.select_related('direction', 'updated_by').order_by('-updated_at')[:8]),
        })

    if can_users:
        values.update({
            'users_total': User.objects.count(),
            'users_individual': User.objects.filter(account_type='individual').count(),
            'users_legal': User.objects.filter(account_type='legal').count(),
            'users_new_today': User.objects.filter(date_joined__date=timezone.localdate()).count(),
        })

    if can_kyc:
        values.update({
            'kyc_pending': KYCVerification.objects.filter(status='pending').count(),
            'kyc_error': KYCVerification.objects.filter(status='error').count(),
            'kyc_approved': KYCVerification.objects.filter(status='approved').count(),
            'kyc_recent': list(KYCVerification.objects.select_related('user').order_by('-created_at')[:8]),
        })

    context.update(values)
    return context

def can_view(permission):
    """Проверка доступа к разделу админки."""
    def check(request):
        user = request.user
        return (
            user.is_authenticated
            and user.is_active
            and user.has_perm(permission)
        )

    return check


def nav(title, icon, model):
    """Создать пункт навигации для модели Django."""
    app_label, model_name = model.lower().split(".")

    return {
        "title": title,
        "icon": icon,
        "link": reverse_lazy(
            f"admin:{app_label}_{model_name}_changelist"
        ),
        "permission": can_view(
            f"{app_label}.view_{model_name}"
        ),
    }


def optional_nav(title, icon, model):
    """
    Показывать пункт только при наличии модели,
    регистрации в admin и прав пользователя.
    """
    app_label, model_name = model.lower().split(".")

    def is_available(request):
        try:
            apps.get_model(app_label, model_name)
            reverse(
                f"admin:{app_label}_{model_name}_changelist"
            )
        except (LookupError, NoReverseMatch):
            return False

        return can_view(
            f"{app_label}.view_{model_name}"
        )(request)

    return {
        "title": title,
        "icon": icon,
        "link": reverse_lazy(
            f"admin:{app_label}_{model_name}_changelist"
        ),
        "permission": is_available,
    }

def configure_admin(namespace):
    """Подключение Django Unfold к настройкам проекта."""

    installed_apps = [
        app
        for app in namespace["INSTALLED_APPS"]
        if app not in (
            "unfold",
            "unfold.contrib.filters",
        )
    ]

    installed_apps.insert(
        0,
        "unfold.contrib.filters",
    )
    installed_apps.insert(
        0,
        "unfold",
    )

    namespace["INSTALLED_APPS"] = installed_apps
    namespace["UNFOLD"] = UNFOLD_CONFIG
